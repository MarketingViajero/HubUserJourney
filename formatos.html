// ID de la carpeta de Google Drive donde se subirán las imágenes const DRIVE_FOLDER_ID = '1RyZTi8HxepFyFcJVuP0tnrufRGBmB3Zf'; // Nombre de la hoja dentro del Sheet const SHEET_NAME = 'UserJourneyData'; // ────────────────────────────────────────────── function jsonResponse(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } // ────────────────────────────────────────────── // ENTRY POINTS // ────────────────────────────────────────────── function doGet(e) { return jsonResponse({ ok: true, message: 'Viajero GAS Backend running' }); } function doPost(e) { try { const payload = JSON.parse(e.postData.contents); const { action } = payload; if (action === 'getAll') return jsonResponse(handleGetAll()); if (action === 'saveField') return jsonResponse(handleSaveField(payload)); if (action === 'saveImage') return jsonResponse(handleSaveImage(payload)); return jsonResponse({ ok: false, error: 'Unknown action' }); } catch(err) { return jsonResponse({ ok: false, error: err.message }); } } // ────────────────────────────────────────────── // HELPERS — Sheet // ────────────────────────────────────────────── function getSheet() { const ss = SpreadsheetApp.openById(SHEET_ID); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { // Crea la hoja si no existe con cabeceras sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow(['cardId', 'field', 'value', 'updatedAt', 'updatedBy']); sheet.setFrozenRows(1); } return sheet; } // Devuelve todos los datos como objeto { cardId: { field: value } } function handleGetAll() { const sheet = getSheet(); const rows = sheet.getDataRange().getValues(); if (rows.length <= 1) return { ok: true, data: {} }; const data = {}; // Recorre de abajo hacia arriba para que el valor más reciente gane for (let i = rows.length - 1; i >= 1; i--) { const [cardId, field, value] = rows[i]; if (!cardId || !field) continue; if (!data[cardId]) data[cardId] = {}; if (data[cardId][field] === undefined) { // ya tiene el más reciente data[cardId][field] = value; } } return { ok: true, data }; } // Guarda o actualiza una fila en el Sheet function handleSaveField(payload) { const { cardId, field, value } = payload; if (!cardId || !field) return { ok: false, error: 'Missing cardId or field' }; const sheet = getSheet(); const rows = sheet.getDataRange().getValues(); const user = Session.getActiveUser().getEmail() || 'anon'; const now = new Date().toISOString(); // Busca si ya existe una fila con ese cardId+field y la actualiza for (let i = 1; i < rows.length; i++) { if (rows[i][0] === cardId && rows[i][1] === field) { sheet.getRange(i + 1, 3, 1, 3).setValues([[value, now, user]]); return { ok: true, updated: true }; } } // Si no existe, agrega una fila nueva sheet.appendRow([cardId, field, value, now, user]); return { ok: true, created: true }; } // ────────────────────────────────────────────── // HELPERS — Drive (imágenes) // ────────────────────────────────────────────── function handleSaveImage(payload) { const { cardId, field, filename, base64 } = payload; if (!cardId || !field || !base64) return { ok: false, error: 'Missing params' }; try { const folder = DriveApp.getFolderById(DRIVE_FOLDER_ID); const blob = Utilities.newBlob(Utilities.base64Decode(base64), 'image/jpeg', filename || (cardId + '_' + field + '.jpg')); // Si ya existe un archivo con ese nombre, lo elimina para reemplazarlo const existing = folder.getFilesByName(blob.getName()); while (existing.hasNext()) { existing.next().setTrashed(true); } const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); // URL directa para usar como src en img tags const fileId = file.getId(); const directUrl = 'https://lh3.googleusercontent.com/d/' + fileId; // Guarda la URL en el Sheet para que otros la puedan cargar handleSaveField({ cardId, field, value: directUrl }); return { ok: true, url: directUrl, fileId }; } catch(err) { return { ok: false, error: err.message }; } }
